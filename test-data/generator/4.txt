<S> -> <Exp>#
<Exp> -> <Exp><Rel><simexp> | <simexp>
<simexp> -> <simexp><PLUSO><simterm> | <simterm>
<simterm> -> <simterm><MULO><term> | <term>
<term> -> +<term> | -<term> | (<Exp>) | not<term> | <ident> | <Number> | true | false
<ident> -> id | <comid> | <idindx> | <callFunc>
<comid> -> id | id.<comid>
<idindx> -> <comid> <listInd>
<listInd> -> [<Exp>] | [<Exp>] <listInd>
<callFunc> -> <comid>(<listExp>)
<listExp> -> <Exp> | <Exp>,<listExp>
<Rel> -> = | != | > | < | <= | >=
<PLUSO> -> + | - | OR
<MULO> -> AND | * | / | mod | div
